class Pointer {
    constructor(n){this.n=BigInt(n);}add(n){return new Pointer(this.n+BigInt(n));}sub(p){return new Pointer(this.n-(p instanceof Pointer?p.n:BigInt(p)));}
    compare(p){return this.n<p.n?-1:this.n>p.n?1:0;}equals(p){return p instanceof Pointer&&this.n===p.n;}isNull(){return this.n===0n;}
    toString(){return '0x'+this.n.toString(16);}valueOf(){return this.n;}
}
